모형 : 평균 전류는 켜져 있는 부품의 대표 평균 전류의 합, 동작 시간 = 0.8 × 용량 ÷ 평균 전류. 송신 · 서보 · 카메라는 순간 전류(피크)가 훨씬 크므로 전압 강하(리셋)도 따로 점검하세요(6번 탭). 질량 한도 350 g 은 「예시」이며 대회마다 다릅니다.

📊 위 : 부품별 질량(누적 막대)과 질량 한도(빨간 선) · 아래 : 부품별 평균 전류(누적 막대)와 필요한 동작 시간에 맞는 한계 전류(초록 선). 선을 넘으면 설계를 고쳐야 합니다.
📐 식으로 정리하기 고등

질량 예산 $M=\sum_k m_k\le M_{\lim}$. 전력 예산 : 평균 전류 $I=\sum_k I_k\eta_k$, 동작 시간 $t=\dfrac{0.8\,C}{I}$. 필요 시간 $t_{\rm need}$ 를 맞추려면 $I\le\dfrac{0.8\,C}{t_{\rm need}}$ — 계산기의 초록 선입니다. 배터리 용량 $C$ 를 올리면 질량 $m_b$ 도 늘어나므로 질량 여유와 동작 시간의 거래가 생깁니다.

교과서 밖 : 제약 조건이 여럿인 설계는 「실행 가능 영역」을 그리는 최적화 문제입니다. 목적함수(예 : 과학 데이터량)를 정하고 질량 · 전력 · 비용 제약 아래서 최대로 만드는 선형계획으로 부품 조합을 찾아볼 수 있습니다.

① BMP280 기압 · 고도 로거 (아두이노 · 100 ms 마다 CSV) 
#include <Wire.h>
#include <Adafruit_BMP280.h>

Adafruit_BMP280 bmp;     // I2C
float P0;                // 방출 직전 지상 기압(hPa) — 기준 기압 (오개념 5 !)

void setup() {
  Serial.begin(115200);
  if (!bmp.begin(0x76)) {                       // 모듈에 따라 0x77 일 수 있음
    Serial.println("BMP280 not found");
    while (1);
  }
  bmp.setSampling(Adafruit_BMP280::MODE_NORMAL,
                  Adafruit_BMP280::SAMPLING_X2,   // 온도
                  Adafruit_BMP280::SAMPLING_X16,  // 기압(잡음 줄이기)
                  Adafruit_BMP280::FILTER_X4,
                  Adafruit_BMP280::STANDBY_MS_1);
  delay(500);
  P0 = bmp.readPressure() / 100.0;              // Pa → hPa
}

void loop() {
  float p = bmp.readPressure() / 100.0;                       // hPa
  float h = 44330.77 * (1.0 - pow(p / P0, 0.190263));         // 상대 고도(m)
  Serial.print(millis()); Serial.print(',');
  Serial.print(p, 2);     Serial.print(',');
  Serial.println(h, 1);
  delay(100);
}
② 자동 전개 규칙 — 「연속 N 회 기준 아래」 (발명 I01) 
// 위 로거 코드에 이어 붙이는 판정 부분 (Servo 라이브러리 필요)
#include <Servo.h>
Servo latch;
const float H_ARM = 400;    // 이 고도를 넘은 적이 있어야 「무장」 — 지상 오작동 방지
const float H_TH  = 300;    // 전개 기준 고도(m)
const int   N     = 5;      // 연속 횟수
int  cnt = 0;
bool armed = false, deployed = false;

// setup() 안 :  latch.attach(9); latch.write(0);   // 0 = 잠금

// loop() 안, h 를 계산한 뒤 :
if (h > H_ARM) armed = true;
if (armed && !deployed) {
  if (h < H_TH) cnt++; else cnt = 0;      // 아래로 읽히면 +1, 한 번이라도 위면 0
  if (cnt >= N) {
    latch.write(90);                       // 걸쇠 풀기
    deployed = true;
  }
}

④ LoRa 수신 — RSSI 로거 (지상국 · 발명 I06 · I08) 
#include <SPI.h>
#include <LoRa.h>

#define LORA_FREQ 920E6     // 사용 지역의 허용 대역 · 출력 규정을 반드시 확인 (국내는 전파 인증 모듈 사용)

void setup() {
  Serial.begin(115200);
  if (!LoRa.begin(LORA_FREQ)) { Serial.println("LoRa fail"); while (1); }
}

void loop() {
  int sz = LoRa.parsePacket();
  if (sz) {
    String s = "";
    while (LoRa.available()) s += (char)LoRa.read();
    Serial.print(millis());            Serial.print(',');
    Serial.print(LoRa.packetRssi());   Serial.print(',');   // dBm
    Serial.print(LoRa.packetSnr());    Serial.print(',');   // dB
    Serial.println(s);
  }
}
⑤ GPS → 목표 방위 · 거리 · 오차각 (아두이노 나노 + SoftwareSerial · 발명 I02 · I07) 
#include <SoftwareSerial.h>
#include <TinyGPSPlus.h>

SoftwareSerial ss(4, 3);          // GPS TX → 4번, RX → 3번
TinyGPSPlus gps;
const double TGT_LAT = 37.0000, TGT_LON = 127.0000;   // 목표 좌표(내 시험장 좌표로 바꾸기)

void setup() { Serial.begin(115200); ss.begin(9600); }

void loop() {
  while (ss.available()) gps.encode(ss.read());
  if (gps.location.isUpdated() && gps.course.isValid()) {
    double brg  = TinyGPSPlus::courseTo(gps.location.lat(), gps.location.lng(), TGT_LAT, TGT_LON);
    double dist = TinyGPSPlus::distanceBetween(gps.location.lat(), gps.location.lng(), TGT_LAT, TGT_LON);
    double err  = brg - gps.course.deg();          // 목표 방위 − 진행 방향
    if (err > 180) err -= 360;                     // −180 ~ +180 으로 감싸기
    if (err < -180) err += 360;
    Serial.print(dist, 1); Serial.print(" m, 오차각 "); Serial.println(err, 0);
  }
}
⑥ 파이썬 분석 — 항력계수 Cd (종합1 · 로그 CSV) 
import numpy as np

g, rho = 9.8, 1.225
# 한 줄 = (질량 kg, 지름 m, 낙하 거리 m, 낙하 시간 s) — 내 측정값으로 바꾸기
data = np.array([[0.020, 0.30, 3.0, 1.61],     # 예시 값 — 내 측정값으로 바꾸기
                 [0.040, 0.30, 3.0, 1.14],
                 [0.080, 0.30, 3.0, 0.80]])
m, D, L, t = data.T
v  = L / t                         # 종단속도(m/s)
x  = m / (np.pi * (D / 2) ** 2)    # m/A (kg/m²)
y  = v ** 2
s  = np.sum(x * y) / np.sum(x * x) # 원점을 지나는 최소제곱 기울기
Cd = 2 * g / (rho * s)
print(f"기울기 = {s:.2f} (m/s)^2/(kg/m^2),  Cd = {Cd:.2f}")
⑦ 파이썬 분석 — 스케일 높이와 평균 기온 (종합2) 
import numpy as np

R, M, g = 8.314, 0.02897, 9.80
# 고도 h(m) 와 기압 P(hPa) — 내 로그로 바꾸기. 첫 줄이 기준(지상)
h = np.array([0, 100, 200, 400, 600, 800, 1000.])
P = np.array([1013.2, 1001.2, 989.3, 965.8, 942.6, 920.0, 897.8])
y = np.log(P[0] / P)              # ln(P0/P) = h / H
slope = np.sum(h * y) / np.sum(h * h)   # 원점을 지나는 기울기 = 1/H
H = 1 / slope                      # 스케일 높이(m)
T = H * M * g / R                  # 평균 기온(K)
print(f"H = {H:.0f} m,  평균 기온 = {T - 273.15:.1f} ℃")
